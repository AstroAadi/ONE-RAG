0742:     ) -> List[Dict[str, Any]]:

0744:         tokens = self.token_counter.get_tokens(text)

0746:             return self._chunk_fixed_size_chars(text, base_metadata)

0748:         stride = max(1, self.target_chunk_tokens - self.chunk_overlap_tokens)
0749: 
0750:         raw: List[Tuple[int, List[int]]] = []
0751:         position = 0
0752:         while position < len(tokens):

0767:             chunk_text = self.token_counter.decode_tokens(toks)
0768:             if chunk_text.strip():
0769:                 chunks.append(self._create_chunk(
0770:                     chunk_text, chunk_index, base_metadata,

0779:     def _chunk_hierarchical(
0780:         self,
0781:         text: str,
0782:         base_metadata: Dict,
0783:         max_parent_tokens: Optional[int] = None,
0784:         max_child_tokens: Optional[int] = None,
0785:         child_overlap_tokens: Optional[int] = None,
0786:     ) -> List[Dict[str, Any]]:
0787:         """
0788:         Two-level hierarchical chunking.
0789: 
0790:         Phase 1 – Create overlapping parent windows over the full token sequence.
0791:         Phase 2 – Split each parent into fixed-size overlapping child windows.
0792: 
0793:         Returns a flat list: parent chunks first, then child chunks in order.
0794: 
0795:         """
0796:         if not text.strip():
0797:             return []
0798: 
0799:         # ── Defaults ─────────────────────────────────────────────────────
0800:         if max_parent_tokens is None:
0801:             max_parent_tokens = max(600, self.target_chunk_tokens * 4)
0802:         if max_child_tokens is None:
0803:             max_child_tokens = self.target_chunk_tokens
0804:         if child_overlap_tokens is None:
0805:             child_overlap_tokens = self.chunk_overlap_tokens
0806: 
0807:         # Safety guards
0808:         max_parent_tokens    = max(max_parent_tokens, max_child_tokens + 1)
0809:         child_overlap_tokens = min(child_overlap_tokens, max_child_tokens // 2)
0810:         parent_overlap_tokens = min(
0811:             int(max_parent_tokens * 0.20), max_parent_tokens // 2
0812:         )
0813: 
0814:         # ── Phase 1: parent windows ───────────────────────────────────────
0815:         all_tokens = self.token_counter.get_tokens(text)
0816:         if not all_tokens:
0817:             return self._chunk_fixed_size_chars(text, base_metadata)
0818: 
0819:         parent_stride = max(1, max_parent_tokens - parent_overlap_tokens)
0820:         parent_records: List[Dict[str, Any]] = []
0821:         p_index  = 1
0822:         position = 0
0823: 
0824:         while position < len(all_tokens):
0825:             end      = min(position + max_parent_tokens, len(all_tokens))
0826:             p_tokens = all_tokens[position:end]
0827:             p_text   = self.token_counter.decode_tokens(p_tokens)
0828:             if p_text.strip():
0829:                 # FIX-4: use hash-based chunk_id (same as _create_chunk does)
0830:                 p_chunk = self._create_chunk(
0831:                     p_text, p_index, base_metadata,
0832:                     position=position, total_tokens=len(all_tokens),
0833:                     chunk_type="parent",
0834:                 )
0835:                 parent_records.append(p_chunk)
0836:                 p_index += 1
0837:             position += parent_stride
0838: 
0839:         # ── Phase 2: children for each parent ────────────────────────────
0840:         all_output: List[Dict[str, Any]] = list(parent_records)   # parents first
0841:         child_counter = 1
0842: 
0843:         for parent in parent_records:
0844:             parent_id   = parent["chunk_id"]
0845:             parent_text = parent["text"]
0846: 
0847:             children = self._split_into_children(
0848:                 parent_text,
0849:                 max_child_tokens,
0850:                 child_overlap_tokens,
0851:                 base_metadata,
0852:                 parent_id=parent_id,
0853:                 child_index_start=child_counter,
0854:             )
0855: 
0856:             # Back-fill child IDs into parent metadata
0857:             parent["metadata"]["child_chunk_ids"] = [c["chunk_id"] for c in children]
0858: 
0859:             all_output.extend(children)
0860:             child_counter += len(children)
0861: 
0862:         n_parents  = len(parent_records)
0863:         n_children = len(all_output) - n_parents
0864:         logger.info(
0865:             f"Hierarchical → {n_parents} parents, {n_children} children  "
0866:             f"(parent={max_parent_tokens}t/{parent_overlap_tokens}t-overlap, "
0867:             f"child={max_child_tokens}t/{child_overlap_tokens}t-overlap)"
0868:         )
0869:         return all_output
